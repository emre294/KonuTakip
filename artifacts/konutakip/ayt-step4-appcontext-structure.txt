 306: function computeStreak(days: string[]): number {
 307:   if (days.length === 0) return 0;
 308:   const sorted = [...days].sort().reverse();
 309:   const today = new Date().toISOString().split("T")[0];
 310:   let streak = 0;
 311:   const current = new Date(today);
 312:   for (let i = 0; i < 365; i++) {
 313:     const dateStr = current.toISOString().split("T")[0];
 314:     if (sorted.includes(dateStr)) {
 315:       streak++;
 316:       current.setDate(current.getDate() - 1);
 317:     } else {
 318:       break;
 319:     }
 320:   }
 321:   return streak;
 322: }
